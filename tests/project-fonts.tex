% Permit file-based font probes, but simulate unavailable system font names.
\RequirePackage{fontspec}
\ExplSyntaxOn
\cs_new_eq:NN \suep_test_font_exists:nTF \IfFontExistsTF
\RenewDocumentCommand \IfFontExistsTF {m m m}
  {\regex_match:nnTF {\.(otf|ttf|ttc|TTF)\Z} {#1}
     {\suep_test_font_exists:nTF {#1}{#2}{#3}}{#3}}
\NewDocumentCommand \SUEPTestFontFile {m}
  {\exp_args:Nee \tl_if_in:nnF {\fontname\font}{\tl_to_str:n {#1}}
     {\errmessage{Expected~project~font~file~#1}}}
\ExplSyntaxOff
